    69→async function fetchText(url: string, timeoutMs: number, accept: string): Promise<string> {
    70→  const controller = new AbortController();
    71→  const timer = setTimeout(() => controller.abort(), timeoutMs);
    72→  try {
    73→    const res = await fetch(url, {
    74→      signal: controller.signal,
    75→      cache: "no-store",
    76→      headers: { "User-Agent": BROWSER_UA, Accept: accept },
    77→      redirect: "follow",
    78→    });
    79→    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    80→    const buffer = await res.arrayBuffer();
    81→    return new TextDecoder("utf-8", { fatal: false }).decode(buffer);
    82→  } finally {
    83→    clearTimeout(timer);
    84→  }
    85→}

    87→function asArray<T>(value: T | T[] | undefined | null): T[] {
    88→  if (value === undefined || value === null) return [];
    89→  return Array.isArray(value) ? value : [value];

    92→function textOf(value: unknown): string {
    93→  if (value === undefined || value === null) return "";
    94→  if (typeof value === "string") return value;
    95→  if (typeof value === "number" || typeof value === "boolean") return String(value);
    96→  if (typeof value === "object") {
    97→    const record = value as Record<string, unknown>;
    98→    if (typeof record["#text"] === "string") return record["#text"];
    99→  }
   100→  return "";
   101→}

   103→function attrOf(value: unknown, attr: string): string {
   104→  if (value && typeof value === "object" && !Array.isArray(value)) {
   105→    const record = value as Record<string, unknown>;
   106→    const raw = record[`@_${attr}`];
   107→    if (typeof raw === "string") return raw;
   108→  }
   109→  return "";
   110→}

   151→function collectCategories(value: unknown): string[] {
   152→  return asArray(value)
   153→    .map((entry) => {
   154→      if (typeof entry === "string") return entry;
   155→      const term = attrOf(entry, "term");
   156→      return term || textOf(entry);
   157→    })
   158→    .map((entry) => decodeEntities(entry).trim())
   159→    .filter(Boolean);

   162→function collectMedia(value: unknown): MediaCandidate[] {
   163→  const out: MediaCandidate[] = [];
   164→  for (const node of asArray(value)) {
   165→    const url = attrOf(node, "url");
   166→    if (!url) continue;
   167→    const widthRaw = parseInt(attrOf(node, "width"), 10);
   168→    out.push({
   169→      url,
   170→      width: Number.isFinite(widthRaw) ? widthRaw : undefined,
   171→    });
   172→  }
   173→  return out;
   174→}

   176→function extractHtmlImages(html: string): MediaCandidate[] {
   177→  if (!html) return [];
   178→  const out: MediaCandidate[] = [];
   179→  const tagRegex = /<img\b[^>]*>/gi;
   180→  let match: RegExpExecArray | null;
   181→  while ((match = tagRegex.exec(html)) !== null) {
   182→    const tag = match[0];
   183→    const src =
   184→      attrSource(tag, "src") ||
   185→      attrSource(tag, "data-src") ||
   186→      attrSource(tag, "data-lazy-src") ||
   187→      attrSource(tag, "data-original");
   188→    if (!src) continue;
   189→    const widthRaw = parseInt(attrSource(tag, "width") ?? "", 10);
   190→    out.push({
   191→      url: decodeEntities(src).trim(),
   192→      width: Number.isFinite(widthRaw) ? widthRaw : undefined,
   193→    });
   194→  }
   195→  return out;
   196→}

   200→  const regex = new RegExp(`\\b${name}\\s*=\\s*["']([^"']+)["']`, "i");
   201→  const match = regex.exec(tag);
   202→  return match ? match[1] : null;

   205→function pickAtomLink(value: unknown): string {
   206→  const links = asArray(value);
   207→  for (const link of links) {
   208→    const rel = attrOf(link, "rel");
   209→    const href = attrOf(link, "href");
   210→    if (!href) continue;
   211→    if (!rel || rel === "alternate") return href;
   212→  }
   213→  return attrOf(links[0], "href");
   214→}

   229→function parseFeedXml(xml: string): ParsedEntry[] {
   230→  const doc = xmlParser.parse(xml) as Record<string, any>;
   231→
   232→  const channel = doc?.rss?.channel;
   233→  const rdfItems = doc?.["rdf:RDF"]?.item;
   234→  const atomEntries = doc?.feed?.entry;
   235→
   236→  const rawItems: any[] = Array.isArray(channel?.item)
   237→    ? channel.item
   238→    : channel?.item
   239→      ? [channel.item]
   240→      : Array.isArray(rdfItems)
   241→        ? rdfItems
   242→        : rdfItems
   243→          ? [rdfItems]
   244→          : Array.isArray(atomEntries)
   245→            ? atomEntries
   246→            : atomEntries
   247→              ? [atomEntries]
   248→              : [];
   249→
   250→  if (rawItems.length === 0) {
   251→    throw new Error("unrecognized feed payload (no items)");
   252→  }
   253→
   254→  return rawItems.map((item) => {
   255→    const contentHtml = textOf(item["content:encoded"]) || textOf(item.content);
   256→    const descriptionHtml = textOf(item.description) || textOf(item.summary);
   257→    const enclosureUrl = attrOf(item.enclosure, "url") || null;
   258→    return {
   259→      guid: textOf(item.guid) || textOf(item.id),
   260→      title: textOf(item.title),
   261→      link: textOf(item.link) || pickAtomLink(item.link) || textOf(item.guid) || textOf(item.id),
   262→      pubDate: textOf(item.pubDate) || textOf(item["dc:date"]) || textOf(item.published) || textOf(item.updated),
   263→      author:
   264→        textOf(item["dc:creator"]) ||
   265→        textOf(item.author) ||
   266→        textOf((item.author as any)?.name),
   267→      categories: collectCategories(item.category),
   268→      contentHtml,
   269→      descriptionHtml,
   270→      media: [...collectMedia(item["media:content"]), ...collectMedia(item["media:thumbnail"])],
   271→      enclosureUrl,
   272→    };
   273→  });
   274→}

   295→/** Tracking pixels, icons, avatars and other junk we refuse to show. */
   296→const BAD_IMG_RE =
   297→  /1x1|pixel|transparent|spacer|blank\.(?:gif|png)|feedburner|feedsportal|feedproxy|avatar|gravatar|badge|rating|sprite|emoji|smiley|\/icon(?:s)?\/|logo|wp-content\/themes|donate|newsletter|captcha|counter|share(?:d)?[-_]/i;
   298→
   299→const GOOD_IMG_RE =
   300→  /\.(?:jpe?g|png|webp)(?:[?#]|$)|\/(?:wp-content\/uploads|wp-json|images?|media|img)\//i;

   302→function scoreImageCandidate(candidate: MediaCandidate): number {
   303→  const url = candidate.url;
   304→  if (!url || url.startsWith("data:") || !/^https?:\/\//i.test(url)) return -100;
   305→  let score = 0;
   306→  if (BAD_IMG_RE.test(url)) score -= 100;
   307→  if (/\.(?:jpe?g|png|webp|gif)(?:[?#]|$)/i.test(url)) score += 3;
   308→  if (GOOD_IMG_RE.test(url)) score += 3;
   309→  if (url.startsWith("https://")) score += 1;
   310→  if (candidate.width !== undefined) {
   311→    if (candidate.width >= 300) score += 5;
   312→    else if (candidate.width >= 200) score += 3;
   313→    else if (candidate.width > 0 && candidate.width <= 80) score -= 40;
   314→  }
   315→  return score;
   316→}

   318→function pickImageFromEntry(entry: {
   319→  mediaCandidates: MediaCandidate[];
   320→  enclosureUrl: string | null;
   321→  contentHtml: string;
   322→  descriptionHtml: string;
   323→}): string | null {
   324→  const pools: MediaCandidate[][] = [
   325→    entry.mediaCandidates,
   326→    entry.enclosureUrl ? [{ url: entry.enclosureUrl }] : [],
   327→    extractHtmlImages(entry.contentHtml),
   328→    extractHtmlImages(entry.descriptionHtml),
   329→  ];
   330→
   331→  for (const pool of pools) {
   332→    let best: { url: string; score: number } | null = null;
   333→    for (const candidate of pool) {
   334→      const score = scoreImageCandidate(candidate);
   335→      if (score > -50 && (!best || score > best.score)) {
   336→        best = { url: candidate.url, score };
   337→      }
   338→    }
   339→    if (best) {
   340→      let url = best.url.trim();
   341→      if (url.startsWith("//")) url = `https:${url}`;
   342→      if (/^https?:\/\//i.test(url)) return url;
   343→    }
   344→  }
   345→  return null;
   346→}

   359→async function fetchOgImage(articleUrl: string): Promise<string | null> {
   360→  try {
   361→    const html = await fetchText(articleUrl, OG_TIMEOUT_MS, "text/html,application/xhtml+xml,*/*");
   362→    for (const pattern of OG_META_PATTERNS) {
   363→      const match = pattern.exec(html);
   364→      if (!match) continue;
   365→      const url = decodeEntities(match[1] ?? "").trim();
   366→      if (/^https?:\/\//i.test(url)) return url;
   367→    }
   368→  } catch {
   369→    // article page unreachable — leave without og image
   370→  }
   371→  return null;
   372→}

   374→async function ogFor(articleUrl: string): Promise<string | null> {
   375→  const cached = ogCache.get(articleUrl);
   376→  if (cached && Date.now() - cached.at < OG_TTL_MS) return cached.image;
   377→  const image = await fetchOgImage(articleUrl);
   378→  ogCache.set(articleUrl, { at: Date.now(), image });
   379→  if (ogCache.size > 800) {
   380→    const oldest = ogCache.keys().next().value;
   381→    if (oldest) ogCache.delete(oldest);
   382→  }
   383→  return image;
   384→}

   386→/** Fill in missing thumbnails by scraping og:image off the article pages. */
   387→async function enrichWithOg(entries: NormalizedEntry[]): Promise<number> {
   388→  const targets = entries.filter((entry) => !entry.imageUrl && /^https?:\/\//i.test(entry.link));
   389→  const selected = targets.slice(0, OG_MAX_PER_CYCLE);
   390→  if (selected.length === 0) return 0;
   391→
   392→  let enriched = 0;
   393→  let cursor = 0;
   394→  const workers = Array.from({ length: Math.min(OG_POOL, selected.length) }, async () => {
   395→    while (cursor < selected.length) {
   396→      const entry = selected[cursor++]!;
   397→      const image = await ogFor(entry.link);
   398→      if (image) {
   399→        entry.imageUrl = image;
   400→        enriched += 1;
   401→      }
   402→    }
   403→  });
   404→  await Promise.all(workers);
   405→  return enriched;
   406→}

   449→function splitSentences(text: string): string[] {
   450→  return (text.match(/[^.!?]+[.!?]+(?:["')\]]+)?(?=\s|$)/g) ?? [text])
   451→    .map((sentence) => sentence.trim())
   452→    .filter(Boolean);

   455→function scoreSentence(sentence: string, index: number): number {
   456→  const lower = sentence.toLowerCase();
   457→  let score = 0;
   458→  for (const [pattern, weight] of WEIGHTED_TERMS) {
   459→    if (pattern.test(lower)) score += weight;
   460→  }
   461→  if (/cve-\d{4}/i.test(lower)) score += 7;
   462→  if (/\d+(?:\.\d+)?\s*(?:%|percent)/.test(lower) || /million|billion|thousand/.test(lower)) {
   463→    score += 4;
   464→  }
   465→  if (lower.length >= 70 && lower.length <= 240) score += 4;
   466→  else if (lower.length < 45) score -= 4;
   467→  if (index < 3) score += 3 - index;
   468→  return score;
   469→}

   471→function extractCves(text: string): string[] {
   472→  const seen = new Set<string>();
   473→  const matches = text.match(/CVE-\d{4}-\d{4,7}/gi) ?? [];
   474→  for (const match of matches) seen.add(match.toUpperCase());
   475→  return [...seen].slice(0, 4);
   476→}

   478→function extractOrgs(text: string): string[] {
   479→  const hits: Array<{ name: string; index: number }> = [];
   480→  for (const { name, regex } of ORG_MATCHERS) {
   481→    const match = regex.exec(text);
   482→    if (match) hits.push({ name, index: match.index });
   483→  }
   484→  hits.sort((a, b) => a.index - b.index);
   485→  const seen = new Set<string>();
   486→  const out: string[] = [];
   487→  for (const hit of hits) {
   488→    const key = hit.name.toLowerCase();
   489→    if (seen.has(key)) continue;
   490→    seen.add(key);
   491→    out.push(hit.name);
   492→    if (out.length >= 4) break;
   493→  }
   494→  return out;
   495→}

   497→function severityFor(text: string, title: string): { score: number; label: IntelSeverityLabel } {
   498→  // title counts double — headline words are strong signals
   499→  const blob = `${title} ${title} ${text}`.toLowerCase();
   500→  let score = 16;
   501→  for (const [pattern, weight] of WEIGHTED_TERMS) {
   502→    if (pattern.test(blob)) score += weight;
   503→  }
   504→  const cveCount = (blob.match(/cve-\d{4}-\d+/g) ?? []).length;
   505→  score += Math.min(10, cveCount * 2);
   506→  const clamped = Math.max(0, Math.min(100, Math.round(score)));
   507→  const label: IntelSeverityLabel =
   508→    clamped >= 72 ? "CRITICAL" : clamped >= 52 ? "HIGH" : clamped >= 32 ? "ELEVATED" : "MODERATE";
   509→  return { score: clamped, label };
   510→}

   512→function buildIntel(
   513→  title: string,
   514→  html: string,
   515→  categoryIds: string[],
   516→): ArticleIntel {
   517→  const text = stripHtml(html);
   518→  const words = text ? text.split(/\s+/).length : 0;
   519→
   520→  // ranked, de-boilerplated sentences
   521→  const sentences = splitSentences(text)
   522→    .filter((sentence) => sentence.length >= 45 && sentence.length <= 340)
   523→    .filter((sentence) => !BOILERPLATE_RE.test(sentence))
   524→    .map((sentence, index) => ({ sentence, index, score: scoreSentence(sentence, index) }))
   525→    .sort((a, b) => b.score - a.score);
   526→
   527→  const ledeRaw = sentences[0]?.sentence ?? (text ? clampText(text, 200) : "");
   528→  const lede = ledeRaw ? clampText(ledeRaw, 220) : "";
   529→
   530→  const highlights: string[] = [];
   531→  for (const candidate of sentences.slice(1)) {
   532→    const line = clampText(candidate.sentence, 170);
   533→    const duplicate = highlights.some(
   534→      (existing) =>
   535→        existing.slice(0, 50).toLowerCase() === line.slice(0, 50).toLowerCase(),
   536→    );
   537→    if (!duplicate) highlights.push(line);
   538→    if (highlights.length >= 2) break;
   539→  }
   540→
   541→  const cves = extractCves(text);
   542→  const orgs = extractOrgs(`${title}. ${text}`);
   543→
   544→  const tags: string[] = [];
   545→  for (const categoryId of categoryIds.slice(0, 2)) {
   546→    const short = CATEGORY_MAP[categoryId]?.short;
   547→    if (short) tags.push(short);
   548→  }
   549→  if (/zero[- ]day|0[- ]day/i.test(text) && !tags.includes("CVE")) tags.push("ZERO-DAY");
   550→  if (/actively exploited|in the wild|known exploited/i.test(text)) tags.push("EXPLOITED");
   551→  if (cves.length > 0 && !tags.includes("CVE")) tags.push("CVE");
   552→
   553→  const severity = severityFor(text, title);
   554→
   555→  return {
   556→    lede,
   557→    highlights,
   558→    cves,
   559→    orgs,
   560→    tags: tags.slice(0, 4),
   561→    severity,
   562→    readMinutes: Math.max(1, Math.round(words / 220)),
   563→  };
   564→}

   568→function entryFromXml(sourceId: string, parsed: ParsedEntry): NormalizedEntry {
   569→  const html = parsed.contentHtml || parsed.descriptionHtml;
   570→  const raw: RawRss2JsonItem = {
   571→    title: parsed.title,
   572→    link: parsed.link,
   573→    guid: parsed.guid,
   574→    pubDate: parsed.pubDate,
   575→    author: parsed.author,
   576→    categories: parsed.categories,
   577→    enclosure: parsed.enclosureUrl ? { link: parsed.enclosureUrl } : null,
   578→    content: html ? clampText(html, 700) : undefined,
   579→    description: parsed.descriptionHtml ? clampText(parsed.descriptionHtml, 400) : undefined,
   580→  };
   581→  return {
   582→    sourceId,
   583→    guid: parsed.guid,
   584→    title: parsed.title,
   585→    link: parsed.link,
   586→    pubDate: parsed.pubDate,
   587→    author: parsed.author,
   588→    categories: parsed.categories,
   589→    contentHtml: parsed.contentHtml,
   590→    descriptionHtml: parsed.descriptionHtml,
   591→    mediaCandidates: parsed.media,
   592→    enclosureUrl: parsed.enclosureUrl,
   593→    imageUrl: null,
   594→    raw,
   595→  };

   598→function entryFromConverter(sourceId: string, item: RawRss2JsonItem): NormalizedEntry {
   599→  const contentHtml = String(item.content ?? "");
   600→  const descriptionHtml = String(item.description ?? "");
   601→  const media: MediaCandidate[] = [];
   602→  if (typeof item.thumbnail === "string" && item.thumbnail) {
   603→    media.push({ url: item.thumbnail });
   604→  }
   605→  if (item.enclosure?.link) {
   606→    media.push({ url: item.enclosure.link });
   607→  }
   608→  return {
   609→    sourceId,
   610→    guid: String(item.guid ?? item.link ?? ""),
   611→    title: String(item.title ?? ""),
   612→    link: String(item.link ?? item.guid ?? ""),
   613→    pubDate: String(item.pubDate ?? ""),
   614→    author: String(item.author ?? ""),
   615→    categories: Array.isArray(item.categories)
   616→      ? item.categories.map((category) => decodeEntities(String(category)).trim()).filter(Boolean)
   617→      : [],
   618→    contentHtml,
   619→    descriptionHtml,
   620→    mediaCandidates: media,
   621→    enclosureUrl: item.enclosure?.link ?? null,
   622→    imageUrl: null,
   623→    raw: item,
   624→  };
   625→}

   627→function mapItem(entry: NormalizedEntry, source: SourceConfig): FeedItem {
   628→  const title = decodeEntities(entry.title).replace(/\s+/g, " ").trim();
   629→  const link = entry.link.trim();
   630→  const html = entry.contentHtml || entry.descriptionHtml;
   631→  const summary = buildExcerpt(html);
   632→  const categories = entry.categories.slice(0, 8);
   633→  const searchBlob = [title, summary, clampText(html, BLOB_CAP), source.name, entry.author, categories.join(" ")]
   634→    .join(" ")
   635→    .toLowerCase();
   636→  const categoryIds = matchCategories(searchBlob);
   637→
   638→  return {
   639→    id: entry.guid || link || `${source.id}:${title}`,
   640→    title,
   641→    pubDate: entry.pubDate,
   642→    publishedAt: parseUtcDate(entry.pubDate),
   643→    link,
   644→    author: decodeEntities(entry.author).trim() || source.name,
   645→    sourceId: source.id,
   646→    sourceName: source.name,
   647→    sourceShort: source.short,
   648→    accent: source.accent,
   649→    summary,
   650→    descriptionHtml: clampText(html, BODY_CAP),
   651→    imageUrl: pickImageFromEntry(entry),
   652→    categories,
   653→    categoryIds,
   654→    searchBlob,
   655→    intel: buildIntel(title, html, categoryIds),
   656→    raw: entry.raw,
   657→  };
   658→}

   678→interface StreamResult {
   679→  source: SourceConfig;
   680→  entries: NormalizedEntry[];
   681→  via: FeedVia;
   682→  error: string | null;
   683→}
   684→
   685→async function fetchStream(source: SourceConfig): Promise<StreamResult> {
   686→  // 1) direct publisher XML — richest payload (media tags, full content)
   687→  try {
   688→    const parsed = await fetchDirectXml(source.rss);
   689→    if (parsed.length > 0) {
   690→      const entries = parsed
   691→        .map((entry) => entryFromXml(source.id, entry))
   692→        .sort((a, b) => parseUtcDate(b.pubDate) - parseUtcDate(a.pubDate))
   693→        .slice(0, PER_SOURCE_CAP);
   694→      return { source, entries, via: "xml", error: null };
   695→    }
   696→  } catch {
   697→    // fall through to the public converter
   698→  }
   699→
   700→  // 2) public rss2json converter (server-side)
   701→  try {
   702→    const rawItems = await fetchViaConverter(source.rss);
   703→    const entries = rawItems
   704→      .map((item) => entryFromConverter(source.id, item))
   705→      .sort((a, b) => parseUtcDate(b.pubDate) - parseUtcDate(a.pubDate))
   706→      .slice(0, PER_SOURCE_CAP);
   707→    return { source, entries, via: "converter", error: null };
   708→  } catch (error) {
   709→    return {
   710→      source,
   711→      entries: [],
   712→      via: "converter",
   713→      error: error instanceof Error ? error.message : "uplink failure",
   714→    };
   715→  }
   716→}
   717→
   718→export async function aggregateNews(): Promise<AggregatedNews> {
   719→  const results = await Promise.all(SOURCES.map(fetchStream));
   720→
   721→  // cross-source dedupe (newest-first ordering decided below)
   722→  const seen = new Set<string>();
   723→  const entries: NormalizedEntry[] = [];
   724→  for (const result of results) {
   725→    for (const entry of result.entries) {
   726→      if (!entry.title.trim() || !/^https?:\/\//i.test(entry.link.trim())) continue;
   727→      const key = normalizeKey(entry.link);
   728→      if (key && seen.has(key)) continue;
   729→      if (key) seen.add(key);
   730→      entries.push(entry);
   731→    }
   732→  }
   733→
   734→  entries.sort((a, b) => parseUtcDate(b.pubDate) - parseUtcDate(a.pubDate));
   735→
   736→  // scrape og:image for articles whose feed shipped no usable media
   737→  const ogEnriched = await enrichWithOg(entries);
   738→
   739→  const sourceMap = Object.fromEntries(SOURCES.map((source) => [source.id, source]));
   740→  const items = entries
   741→    .map((entry) => {
   742→      const source = sourceMap[entry.sourceId];
   743→      return source ? mapItem(entry, source) : null;
   744→    })
   745→    .filter((item): item is FeedItem => item !== null);
   746→
   747→  const sources: SourceOutcome[] = results.map((result) => ({
   748→    id: result.source.id,
   749→    name: result.source.name,
   750→    status: result.error ? "offline" : result.entries.length > 0 ? "online" : "empty",
   751→    via: result.via,
   752→    count: result.entries.length,
   753→    error: result.error ?? undefined,
   754→  }));
   755→
   756→  return { generatedAt: Date.now(), items, sources, ogEnriched };
   757→}

   759→/* ── module cache: fresh 4 min, stale-grace 30 min ────────────────── */
   760→
   761→let cache: { at: number; payload: AggregatedNews } | null = null;
   762→let inflight: Promise<AggregatedNews> | null = null;
   763→
   764→export interface NewsSnapshot {
   765→  payload: AggregatedNews;
   766→  ageMs: number;
   767→  fresh: boolean;
   768→}
   769→
   770→export async function getNewsSnapshot(): Promise<NewsSnapshot> {
   771→  const now = Date.now();
   772→
   773→  if (cache && now - cache.at < CACHE_TTL_MS) {
   774→    return { payload: cache.payload, ageMs: now - cache.at, fresh: true };
   775→  }
   776→
   777→  if (cache && now - cache.at < CACHE_STALE_MS) {
   778→    // serve stale instantly, refresh in the background
   779→    if (!inflight) {
   780→      const task = aggregateNews();
   781→      inflight = task;
   782→      void task
   783→        .then((payload) => {
   784→          cache = { at: Date.now(), payload };
   785→        })
   786→        .catch(() => {
   787→          // background refresh failed — keep serving stale
   788→        })
   789→        .finally(() => {
   790→          inflight = null;
   791→        });
   792→    }
   793→    return { payload: cache.payload, ageMs: now - cache.at, fresh: false };
   794→  }
   795→
   796→  if (!inflight) {
   797→    inflight = aggregateNews()
   798→      .then((payload) => {
   799→        cache = { at: Date.now(), payload };
   800→        inflight = null;
   801→        return payload;
   802→      })
   803→      .catch((error) => {
   804→        inflight = null;
   805→        throw error;
   806→      });
   807→  }
   808→
   809→  const payload = await inflight;
   810→  return { payload, ageMs: 0, fresh: true };
   811→}
